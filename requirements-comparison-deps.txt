# Exact InsightFace dependencies for the optional Experiment 8 comparison.
#
# insightface declares opencv-python as a dependency. Installing it normally
# resolves that distribution alongside opencv-python-headless; both ship the
# same cv2 module, so whichever wins the import shadows the other and silently
# changes detection and embedding numerics. insightface is therefore installed
# with --no-deps and its dependencies supplied here explicitly, with the
# headless build pinned.
#
# Versions recorded from the environment that produced the published
# Experiment 8 results. Install via scripts/install_comparison_environment.sh,
# which verifies the outcome rather than assuming it.
onnxruntime==1.28.0
onnx==1.22.0
scipy==1.18.0
scikit-image==0.26.0
tqdm==4.70.0
requests==2.34.2

# Required by opencv-python-headless and by insightface's array handling; kept
# identical to the core project pin so one NumPy is effective.
numpy==2.5.1

# The only OpenCV that may be present. opencv-python must NOT be installed.
opencv-python-headless==4.13.0.92
